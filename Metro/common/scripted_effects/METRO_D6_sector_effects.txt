# Д-6 sector excavation effects
# Excavation rates per level (progress out of 100):
#   Level 1 (S1):        +2.0/day  (~50 days)
#   Level 2 (S2-S6):     +1.0/day  (~100 days)
#   Level 3 (S7-S9):     +0.5/day  (~200 days)
#   Level 4 (S10):       +0.25/day (~400 days)

d6_start_sector_1 = {
	custom_effect_tooltip = D6_START_SECTOR_L1_tt
	hidden_effect = {
		if = {
			limit = { d6_can_start_excavation = yes }
			set_variable = { global.d6_active_sector = 1 }
			set_variable = { global.d6_sector_progress = 0 }
		}
	}
}

# Called each day for the country that owns and controls state 106.
d6_excavation_daily_tick = {
	if = {
		limit = {
			check_variable = { global.d6_active_sector > 0 }
		}

		# Determine the daily rate from the active sector's level
		if = {
			limit = { check_variable = { global.d6_active_sector = 1 } }
			set_temp_variable = { d6_rate_temp = 2 }
		}
		else_if = {
			limit = {
				check_variable = { global.d6_active_sector > 1 }
				check_variable = { global.d6_active_sector < 7 }
			}
			set_temp_variable = { d6_rate_temp = 1 }
		}
		else_if = {
			limit = {
				check_variable = { global.d6_active_sector > 6 }
				check_variable = { global.d6_active_sector < 10 }
			}
			set_temp_variable = { d6_rate_temp = 0.5 }
		}
		else = {
			set_temp_variable = { d6_rate_temp = 0.25 }
		}

		add_to_variable = { global.d6_sector_progress = d6_rate_temp }

		if = {
			limit = { check_variable = { global.d6_sector_progress > 100 } }
			set_variable = { global.d6_sector_progress = 100 }
		}

		# Completion: flag the sector as cleared and fire its event
		if = {
			limit = { check_variable = { global.d6_sector_progress > 99.899 } }

			if = {
				limit = { check_variable = { global.d6_active_sector = 1 } }
				set_global_flag = D6_sector_1_cleared
				country_event = { id = metro_d6_sector.1 }
			}
			else_if = {
				limit = { check_variable = { global.d6_active_sector = 2 } }
				set_global_flag = D6_sector_2_cleared
				country_event = { id = metro_d6_sector.2 }
			}
			else_if = {
				limit = { check_variable = { global.d6_active_sector = 3 } }
				set_global_flag = D6_sector_3_cleared
				country_event = { id = metro_d6_sector.3 }
			}
			else_if = {
				limit = { check_variable = { global.d6_active_sector = 4 } }
				set_global_flag = D6_sector_4_cleared
				country_event = { id = metro_d6_sector.4 }
			}
			else_if = {
				limit = { check_variable = { global.d6_active_sector = 5 } }
				set_global_flag = D6_sector_5_cleared
				country_event = { id = metro_d6_sector.5 }
			}
			else_if = {
				limit = { check_variable = { global.d6_active_sector = 6 } }
				set_global_flag = D6_sector_6_cleared
				country_event = { id = metro_d6_sector.6 }
			}
			else_if = {
				limit = { check_variable = { global.d6_active_sector = 7 } }
				set_global_flag = D6_sector_7_cleared
				country_event = { id = metro_d6_sector.7 }
			}
			else_if = {
				limit = { check_variable = { global.d6_active_sector = 8 } }
				set_global_flag = D6_sector_8_cleared
				country_event = { id = metro_d6_sector.8 }
			}
			else_if = {
				limit = { check_variable = { global.d6_active_sector = 9 } }
				set_global_flag = D6_sector_9_cleared
				country_event = { id = metro_d6_sector.9 }
			}
			else_if = {
				limit = { check_variable = { global.d6_active_sector = 10 } }
				set_global_flag = D6_sector_10_cleared
				country_event = { id = metro_d6_sector.10 }
			}

			set_variable = { global.d6_active_sector = 0 }
			set_variable = { global.d6_sector_progress = 0 }
		}
	}
}
